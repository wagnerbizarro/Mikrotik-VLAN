set [ find default-name=ether2 ] comment=SW1-eth2

/interface wireless security-profiles

add bridge=bridge-vlan12 interface=ether1 pvid=12

/interface bridge vlan
add bridge=*5 tagged=ether1,ether2 vlan-ids=12
